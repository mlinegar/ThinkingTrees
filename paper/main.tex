\appendix

\noindent This appendix provides complete technical foundations, detailed proofs, and practical guidance for the Oracle-Preserving Summarization framework. It is organized to support both theoretical understanding and empirical deployment.

\medskip

% Mathematical Foundations
\noindent\textbf{Appendix~\ref{app:notation}: Notation Recap} collects all notation and definitions.
\noindent\textbf{Appendix~\ref{app:proofs-core}: Proofs of Main Results} establishes the core guarantees.

% Learning Theory
\noindent\textbf{Appendix~\ref{app:blackwell}: Score Transport} shows how supervision factors through the oracle.
\noindent\textbf{Appendix~\ref{sec:dpo}: DPO Bounds} proves DPO equivalence and gap bounds.
\noindent\textbf{Appendix~\ref{sec:grpo}: GRPO Bounds} extends to Plackett-Luce and DeepSeek-R1 style training.
\noindent\textbf{Appendix~\ref{app:preference-proofs}: Preference Learning Proofs} provides detailed Lipschitz and equivalence proofs.
\noindent\textbf{Appendix~\ref{app:adaptive-tree-proofs}: Adaptive Tree Objective Proofs} gives full proofs for expected-tree optimizer transfer under oracle uncertainty.
\noindent\textbf{Appendix~\ref{app:theorem-backed}: Theorem-Backed Reductions} explains the broadest exact and approximate reduction interface, the codec route, and the collapse to the global $A1/A2/A3$ regime.

% Practical Guidance
\noindent\textbf{Appendix~\ref{app:practical_audit}: Practical Audit} provides step-by-step procedures.
\noindent\textbf{Appendix~\ref{app:confidence}: Confidence Intervals} gives Hoeffding bounds and sample size calculations.
\noindent\textbf{Appendix~\ref{app:clustered-inference}: Clustered and Calibrated Inference} explains how clustered confidence intervals and calibration events feed directly into the final DSL bound object.
\noindent\textbf{Appendix~\ref{app:measurement-error}: Measurement Error and Proxy-Aware Sampling} gives the transport-plus-measurement decomposition and the exact residual formula for nonclassical IPW mismatch.
\noindent\textbf{Appendix~\ref{app:measure-audit}: Measure-Theoretic Audit} formalizes concentration inequalities.
\noindent\textbf{Appendix~\ref{app:sampling}: Sampling Theory} covers IPW estimators and stratification.
\noindent\textbf{Appendix~\ref{app:bootstrap}: Consistency Bootstrap} describes the DSPy optimization loop.

% Alternative Formulations
\noindent\textbf{Appendix~\ref{sec:global}: Global Variant} presents an algebraic formulation.
\noindent\textbf{Appendix~\ref{app:mergeable}: Mergeable Relationship} connects to database theory.
\noindent\textbf{Appendix~\ref{sec:info-sieve}: Information-Theoretic View} offers an optional perspective.

% Existence, Examples, and Necessity
\noindent\textbf{Appendix~\ref{app:existence}: Existence} establishes that oracle-preserving summarizers exist.
\noindent\textbf{Appendix~\ref{sec:examples}: Worked Examples} instantiates the framework for concrete tasks.
\noindent\textbf{Appendix~\ref{app:pipeline}: Training Pipeline} analyzes multi-stage gap composition.
\noindent\textbf{Appendix~\ref{app:counterexample}: Counterexample} proves L3 (Idempotence) is independent.

% NEW: Formal Verification
