\newcommand{\HierarchyBackdrop}{%
  \foreach \i [evaluate=\i as \xx using {1.2 + 2.15*(\i-1)}] in {1,...,8}{
    \node[rawblock] (x\i) at (\xx,0) {$x_{\i}$};
    \node[sumnode]  (g\i) at (\xx,1.35) {$s_{\i}=g(x_{\i})$};
    \draw[pipeline] (x\i) -- (g\i);
  }
  \foreach \a/\b in {1/2,3/4,5/6,7/8}{
    \node[mergeop] (p\a\b) at ($ (g\a)!0.5!(g\b) + (0,1.25) $) {$\concat$};
    \node[sumnode] (s\a\b) at ($ (p\a\b) + (0,1.25) $) {$s_{\a\b}=g(s_{\a} \concat s_{\b})$};
    \draw[pipeline] (g\a) -- (p\a\b);
    \draw[pipeline] (g\b) -- (p\a\b);
    \draw[pipeline] (p\a\b) -- (s\a\b);
  }
  \foreach \L/\R in {12/34,56/78}{
    \node[mergeop] (p\L\R) at ($ (s\L)!0.5!(s\R) + (0,1.25) $) {$\concat$};
    \node[sumnode] (s\L\R) at ($ (p\L\R) + (0,1.25) $) {$s_{\L\R}=g(s_{\L} \concat s_{\R})$};
    \draw[pipeline] (s\L) -- (p\L\R);
    \draw[pipeline] (s\R) -- (p\L\R);
    \draw[pipeline] (p\L\R) -- (s\L\R);
  }
  \node[mergeop] (proot) at ($ (s1234)!0.5!(s5678) + (0,1.25) $) {$\concat$};
  \node[sumnode] (sroot) at ($(proot) + (0,1.25)$) {$s_{\text{root}}=g(s_{1234} \concat s_{5678})$};
  \draw[pipeline] (s1234) -- (proot);
  \draw[pipeline] (s5678) -- (proot);
  \draw[pipeline] (proot) -- (sroot);
}

\newcommand{\SufficiencyOverlay}{%
  \node[suff node] (suffRaw) at (x1) {$x_{1}$};
  \node[suff node] (suffSum) at (g1) {$s_{1}$};
  \draw[suff arrow] (suffRaw) -- (suffSum);

  \node[comparator, draw=suffColor, text=suffColor] (suffComp) at ($(x1)!0.5!(g1) + (-2.1,0.2)$) {$\fstar$\\agree?};
  \draw[suff arrow] (suffRaw) -- (suffComp);
  \draw[suff arrow] (suffSum) -- (suffComp);

  \node[law title, text=suffColor] (suffTitle) at ($(suffComp)+(0,1.15)$) {(A) Sufficiency};
  \node[law desc, text=suffColor]  (suffDesc)  at ($(suffComp)+(0,-1.05)$) {Re-read a raw block and confirm $g$ keeps the oracle label: $\fstar(x_1) = \fstar(s_1)$.};

  \begin{scope}[on background layer]
    \node[suff box, fit=(suffRaw)(suffSum)(suffComp)(suffTitle)(suffDesc)] {};
  \end{scope}
}

\newcommand{\MergeOverlay}{%
  \node[merge node] (mergeRawSeven) at (x7) {$x_{7}$};
  \node[merge node] (mergeRawEight) at (x8) {$x_{8}$};
  \node[merge node] (mergeSumSeven) at (g7) {$s_{7}$};
  \node[merge node] (mergeSumEight) at (g8) {$s_{8}$};
  \node[merge node] (mergeTree) at (s78) {$s_{78}$};

  \draw[merge arrow] (mergeRawSeven) -- (mergeSumSeven);
  \draw[merge arrow] (mergeRawEight) -- (mergeSumEight);
  \draw[merge arrow] (mergeSumSeven) -- (p78);
  \draw[merge arrow] (mergeSumEight) -- (p78);
  \draw[merge arrow] (p78) -- (mergeTree);

  \node[merge node, align=center, text width=2.9cm] (mergeJoint) at ($(s78) + (3.4,0)$) {$s^{\text{joint}}_{78}=g(x_7 \concat x_8)$};
  \draw[merge arrow, dashed] ($(mergeRawSeven.east)!0.5!(mergeRawEight.east)$) to[out=15,in=-160] (mergeJoint.south);

  \node[comparator, draw=mergeColor, text=mergeColor] (mergeComp) at ($(mergeTree)!0.5!(mergeJoint) + (0,1.2)$) {$d_{\mathcal{Y}} \approx 0?$};
  \draw[merge arrow] (mergeTree) -- (mergeComp);
  \draw[merge arrow] (mergeJoint) -- (mergeComp);

  \node[law title, text=mergeColor] (mergeTitle) at ($(mergeComp)+(0,1.2)$) {(B) Merge Consistency};
  \node[law desc, text=mergeColor]  (mergeDesc)  at ($(mergeComp)+(0,-1.1)$) {Check whether the tree merge $s_{78}$ agrees with a single call on $x_7 \concat x_8$.};

  \begin{scope}[on background layer]
    \node[merge box, fit=(mergeRawSeven)(mergeRawEight)(mergeSumSeven)(mergeSumEight)(mergeTree)(mergeJoint)(mergeComp)(mergeTitle)(mergeDesc)] {};
  \end{scope}
}

\newcommand{\IdempotenceOverlay}{%
  \node[idemp node] (idempRoot) at (sroot) {$s_{\text{root}}$};
  \node[idemp node] (idempRepeat) at ($(sroot) + (0,1.9)$) {$g(s_{\text{root}})$};
  \draw[idemp arrow, dashed] (idempRoot) -- node[right, font=\scriptsize, text=idempColor] {rerun $g$} (idempRepeat);

  \node[comparator, draw=idempColor, text=idempColor] (idempComp) at ($(idempRoot)+(0,-1.2)$) {$d_{\mathcal{Y}} \approx 0?$};
  \draw[idemp arrow] (idempRoot) -- (idempComp);
  \draw[idemp arrow] (idempRepeat) -- (idempComp);

  \node[law title, text=idempColor] (idempTitle) at ($(idempRepeat)+(0,0.9)$) {(C) Idempotence};
  \node[law desc, text=idempColor]  (idempDesc)  at ($(idempComp)+(0,-1.1)$) {Summaries should be stable: re-summarizing $s_{\text{root}}$ must not change $\fstar$.};

  \begin{scope}[on background layer]
    \node[idemp box, fit=(idempRoot)(idempRepeat)(idempComp)(idempTitle)(idempDesc)] {};
  \end{scope}
}

\newcommand{\HierarchyFrame}[4]{%
  \begin{tikzpicture}[x=1.05cm, y=1.28cm, >=Latex, font=\sffamily]
    \HierarchyBackdrop
    \node[scene label, anchor=west] at (-1.2,5.9) {#1};
    \ifnum#2=1\relax
      \SufficiencyOverlay
    \fi
    \ifnum#3=1\relax
      \MergeOverlay
    \fi
    \ifnum#4=1\relax
      \IdempotenceOverlay
    \fi
  \end{tikzpicture}
}
