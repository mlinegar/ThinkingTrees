% Helper: render one grid panel with its sampled-node set.
% \gridpanel{<comma-list>} opens a TeX group, sets \samplednodes locally,
% \input's the shared tikz, and scales the result to the panel width.
\newcommand{\gridpanel}[1]{%
  \begingroup
    \renewcommand{\samplednodes}{#1}%
    \resizebox{4.6cm}{!}{\input{01_base_plain.tikz}}%
  \endgroup
}
